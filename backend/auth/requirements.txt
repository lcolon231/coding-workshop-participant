# Runtime dependencies for the auth Lambda. Exact == pins: infra/lambda.tf sets
# build_in_docker = false, so these resolve against the deploy host's Python.
# That host is the VDI (linux-x86_64, CPython 3.13), an exact match for the
# python3.13 runtime in infra/locals.tf:56, so manylinux wheels apply cleanly.

SQLAlchemy==2.0.54
# The repo example pins psycopg[binary]==3.2.3. Taking a current patch release
# instead: the "proven in this pipeline" property is really just cp313 wheel
# availability, which 3.3.x also has, and a two-year-old pin forgoes security
# fixes for no benefit. Same driver, same URL scheme (postgresql+psycopg://).
psycopg[binary]==3.3.6

# Chosen over a hand-rolled router: Pydantic v2 request validation, the
# details[] array straight from the validation error, and OpenAPI docs are all
# rubric-scored. SQLAlchemy is the larger memory line item; if the 128 MB
# ceiling bites, drop the ORM before dropping this.
fastapi==0.141.1

# starlette is fastapi's transitive dependency, constrained only as >=0.46.0.
# Left unpinned, a fresh install resolves 1.x, whose TestClient deprecates
# httpx in favour of httpx2 and, under filterwarnings=error, fails the suite
# at collection (first CI run). 0.49.3 is the version the suite is verified
# against; the same pin as the other service, on purpose.
starlette==0.49.3

# A runtime dependency, not just a dev tool: Aurora is not publicly accessible
# (infra/rds.tf sets no publicly_accessible), so the only compute that can
# migrate it is a Lambda inside the VPC. Imported lazily in admin_actions so it
# costs zip bytes rather than cold-start memory on the request path.
alembic==1.20.0

# bcrypt directly, never passlib: passlib 1.7.4 (the last release, 2020) reads
# bcrypt.__about__.__version__, which bcrypt >= 4.1 removed, producing a
# traceback-in-a-warning on every hash.
bcrypt==5.0.0
# PyJWT rather than python-jose (unmaintained, CVE history). HS256 needs no
# `cryptography` wheel, saving ~10 MB unzipped and an arch risk.
PyJWT==2.11.0

# Required by Pydantic's EmailStr. Pinned explicitly rather than relying on the
# pydantic[email] extra, so the deploy-time install resolves the same version.
email-validator==2.3.0

# ASGI -> Lambda adapter. Imported lazily in function.py, only for HTTP events.
mangum==0.22.0
